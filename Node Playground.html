    The server code
    
        
    Node's built-in http module. You write the routing yourself.
    

    const http = require("http");

http.createServer((req, res) => {
  const url = new URL(req.url, "http://localhost");

  if (req.method === "GET" && url.pathname === "/time") {
    res.writeHead(200, { "Content-Type": "application/json" });
    return res.end(JSON.stringify({ now: new Date() }));
  }
  if (req.method === "GET" && url.pathname === "/greet") {
    const name = url.searchParams.get("name") || "stranger";
    // ...respond with { message: `Hello, ${name}!` }
  }
  if (req.method === "POST" && url.pathname === "/echo") {
    let body = "";
    req.on("data", (chunk) => (body += chunk));
    req.on("end", () => { /* respond with body */ });
    return;
  }
  res.writeHead(404);
  res.end("Not found");
}).listen(3001);
